def place_ships(n,av):
    og_num_ship_po=n
    q=av.copy()
    while n!=0:
        n=og_num_ship_po
        l=[]
        po=random.choice(av)
        av.remove(po)
        n-=1
        l.append(po)
        r=random.choice(['a','b','c','d'])
        if r=='a' or r=='b':
            a=1
            b=1
            while ([po[0], po[1] - a] in av or [po[0], po[1] + b] in av) and n != 0:
                t=random.choice(['a','b'])
                if t=='a':
                    g=[po[0],po[1]-a]
                    if g in av:
                        l.append(g)
                        av.remove(g)
                        n-=1
                        a+=1
                else:
                    g = [po[0], po[1]+b]
                    if g in av:
                        l.append(g)
                        av.remove(g)
                        n-=1
                        b += 1
            if n!=0:
                l=[po]
                av=q.copy()
                n=(og_num_ship_po)-1
            c = 1
            d = 1
            while ([po[0] - c, po[1]] in av or [po[0] + d, po[1]] in av) and n != 0:

                if t == 'c':

                        av.remove(g)
